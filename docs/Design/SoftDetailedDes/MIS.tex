\section{Introduction}

The following document details the Module Interface Specifications for \progname{}.
This program implements the Cut Edge Weight Statistic measurement, proposed by Zighed et al. \cite{zighed_statistical_2005}. 
Complementary documents include the System Requirement Specifications
and Module Guide.  The full documentation and implementation can be
found at \href{https://github.com/LesleyWheat/CAS741_CEWS}{GitHub}.

\section{Notation}

\mycomment{You should describe your notation.  You can use what is below as
  a starting point.}

The base structure of the MIS for modules comes from \cite{HoffmanAndStrooper1995},
with the addition that template modules have been adapted from
\cite{GhezziEtAl2003} and the addition of descriptions for readability.
Note that the symbol := is used for assignment.
\mycomment{The mathematical notation comes from Chapter 3 of
\citet{HoffmanAndStrooper1995}.  
For instance, the symbol := is used for a
multiple assignment statement and conditional rules follow the form $(c_1
\Rightarrow r_1 | c_2 \Rightarrow r_2 | ... | c_n \Rightarrow r_n )$.}

The following table summarizes the primitive data types used by \progname.

\noindent
The specification of \progname \ uses some derived data types: sequences, strings, and
tuples. Sequences are lists filled with elements of the same data type. Strings
are sequences of characters. Tuples contain a list of values, potentially of
different types. In addition, \progname \ uses functions, which
are defined by the data types of their inputs and outputs. Local functions are
described by giving their type signature followed by their specification.

\hline
processInput & \makecell[1]{inData: $\mathbb{M}_{n \times m}$, \\ inLabels: $\mathbb{M}_{n \times 1}$} & \makecell[1]{data: $\mathbb{M}_{n \times m}$, \\ labels: $\mathbb{M}_{n \times 1}$} & \makecell[1]{typeError \\ valueError} \\
\hline
\end{tabular}
\end{center}

\subsection{Semantics}

\subsubsection{State Variables}
None.

\subsubsection{Environment Variables}
None.

\subsubsection{Assumptions}
\begin{itemize}
\item The input data has been formatted correctly with respect to association and orientation (for example, that inData[i] is the data point associated with inLabels[i]).
\end{itemize}

\subsubsection{Access Routine Semantics}

\noindent processInput(inData, inLabels):
\begin{itemize}
\item Description: Verify and transform input if applicable.
\begin{itemize}
\item Check if input is a correct data type
\item Check input meets constraints
\item Convert information to internal data programming object
\end{itemize}
\item Transition: None
\item Output: $out:=$
\begin{itemize}
\item data: the inputted data converted to the correct internal data storage type (from matrix to Python Numpy matrix) for use by the Calculation Module \ref{M_CA}
\item labels: the inputted labels converted to the correct internal data storage type (from matrix to Python Numpy matrix) for use by the Calculation Module \ref{M_CA}
\end{itemize}
\item Exceptions: $exc:=$
    \begin{itemize}
    \item typeError: 
    \begin{itemize}
    \item inData or inLabels is not an accepted data object (for example, a string not a matrix).
    \item inData or inLabels contain non-numeric values.
    \item Different rows of inData have a different $m$ value.
    \item Any index in inData or inLabels is empty or null.
    \end{itemize}
    \item valueError:
        \begin{itemize}
        \item inData or inLabels do not have the same value of $n$.
        \item There is only one class label used (a minimum of two classes must exist in the dataset).
        \item The inLabels is not of size $n \times 1$.
        \end{itemize}
\end{itemize}
\end{itemize}


\subsubsection{Local Functions}
None.

\subsection{Semantics}
\subsubsection{State Variables}
None.

\subsubsection{Environment Variables}
None.

\subsubsection{Assumptions}
\begin{itemize}
\item Input variables have been processed through Input Module \ref{M_I}.
\end{itemize}

\subsubsection{Access Routine Semantics}

\noindent calcCEWS(data, labels):
\begin{itemize}
\item Description: Preform the algorithm from \cite{zighed_statistical_2005}.
\item Transition: None.
\item Output: $out:=$
\begin{itemize}
\item The Cut Edge Weight Statistic: $J_{N} := \frac {J}{I+J}$ where:
\begin{itemize}
\item $J = \sum_{r=1}^{k-1} \sum_{s=r+1}^{k} J_{r,s};$ $ 0 \leq J \leq n^2$
\item $J_{r,s}  = \sum^{n}_{i=1} \sum^n_{j=1} V_{ij} I(\mbox{labels}_i = r \And \mbox{labels}_j = s)$ 
\item $I = \sum_{r=1}^{k} I_r, 0\leq I \leq  n^2$ 
\item $I_{r}  = \sum^{n-1}_{i=1} \sum^n_{j=i+1} V_{ij} I(\mbox{labels}_i = r \And \mbox{labels}_j = r)$. 
\item $V$ is obtained from the Graph Module \ref{M_G} for the data input.
\end{itemize}
\end{itemize}
\item Exceptions: None.
\end{itemize}


\subsubsection{Local Functions}
None.

\subsubsection{Assumptions}
\begin{itemize}
\item $\mbox{data}$ has been processed by Input Module \ref{M_I}.
\end{itemize}

\subsubsection{Access Routine Semantics}

\noindent relativeNeighbourGraph(data):

\begin{itemize}
\item Description: Construct a relative neighbour graph.
\item Transition: None.
\item Output: $out:=$
\begin{itemize}
\item Connection matrix $V := (v_{ij}), i = 1,2, ... ,n; j = 1,2, ... , n;$ \\ 
where $v_{ij} :=$ checkEdge\_RNG($\mbox{distanceMatrix}$, $i$, $j$) given: \\ $\mbox{distanceMatrix}_{ij} = d(data_i, data_j)$ where $d(a,b)$ is Euclidean Distance.
\end{itemize}
\item Exceptions: None.
\end{itemize}


\subsubsection{Local Functions}
\noindent checkEdge\_RNG($\mbox{distanceMatrix}$, $a$, $b$):
\begin{itemize}
\item Description: Checks if an edge exists between points $i$ and $j$.
\item Inputs:
\begin{itemize}
\item $\mbox{distanceMatrix}$: a distance matrix, $\mathbb{M}$, size of $n \times n$
\item $a$: $\mathbb{N}$, $1 \leq a \leq n$
\item $b$: $\mathbb{N}$, $1 \leq b \leq n$
\end{itemize}
\item Transition: None.
\item Output: $out:=$
\begin{itemize}
\item Edge existence:
$E := I([\mbox{distanceMatrix}_{ab} \leq \max(\mbox{distanceMatrix}_{ac}, \mbox{distanceMatrix}_{bc}) \forall c $ where $1 \leq c \leq n, c \neq a, c \neq b ]$).
\end{itemize}
\item Exceptions: None.
\end{itemize}
